+-- UI9A. Read cooldown APIs; never replace protected spell/action functions.
+local tip = GameTooltip
+if not tip or not hooksecurefunc then return end
+local query, elapsedSinceUpdate
+local ceil, floor, max = math.ceil, math.floor, math.max
+local prefixes = {}
+for _, key in ipairs({"ITEM_COOLDOWN_TIME", "ITEM_COOLDOWN_TIME_SEC", "ITEM_COOLDOWN_TIME_MIN", "ITEM_COOLDOWN_TIME_HOURS", "ITEM_COOLDOWN_TIME_DAYS"}) do
+    local value = _G[key]
+    if type(value) == "string" then
+        local at = value:find("%", 1, true)
+        if at and at > 1 then prefixes[#prefixes + 1] = value:sub(1, at - 1) end
+    end
+end
+local function IsRemainingLine(text)
+    if type(text) ~= "string" then return false end
+    for _, prefix in ipairs(prefixes) do
+        if text:sub(1, #prefix) == prefix then return true end
+    end
+    return false
+end
+local function Format(seconds)
+    seconds = ceil(max(0, seconds))
+    if seconds >= 3600 then
+        return string.format("%d:%02d:%02d", floor(seconds / 3600), floor(seconds / 60) % 60, seconds % 60)
+    elseif seconds >= 60 then
+        return string.format("%d:%02d", floor(seconds / 60), seconds % 60)
+    end
+    return tostring(seconds) .. "s"
+end
+local function Refresh()
+    if not query or not tip:IsShown() then return end
+    local start, duration, enabled = query()
+    if type(start) ~= "number" or type(duration) ~= "number" or enabled == 0 then return end
+    local remaining = max(0, start + duration - GetTime())
+    if start == 0 or duration == 0 then remaining = 0 end
+    local name = tip:GetName()
+    if not name then return end
+    for i = 1, tip:NumLines() do
+        local line = _G[name .. "TextLeft" .. i]
+        if line and IsRemainingLine(line:GetText()) then
+            -- Only the native remaining line: preserve base cooldown, costs and descriptions.
+            line:SetText(remaining > 0 and string.format(ITEM_COOLDOWN_TIME, Format(remaining)) or "")
+        end
+    end
+end
+local function TrackSpell(id)
+    local name = id and GetSpellInfo(id)
+    if not name then return end
+    query = function() return GetSpellCooldown(name) end
+    Refresh()
+end
+tip:HookScript("OnTooltipCleared", function() query = nil; elapsedSinceUpdate = 0 end)
+tip:HookScript("OnHide", function() query = nil end)
+tip:HookScript("OnTooltipSetSpell", function(self)
+    local _, id = self:GetSpell()
+    TrackSpell(id)
+end)
+if tip.SetAction then
+    hooksecurefunc(tip, "SetAction", function(self, slot)
+        query = function() return GetActionCooldown(slot) end
+        Refresh()
+    end)
+end
+if tip.SetSpell then
+    hooksecurefunc(tip, "SetSpell", function(self, slot, book)
+        query = function() return GetSpellCooldown(slot, book) end
+        Refresh()
+    end)
+end
+if tip.SetHyperlink then
+    hooksecurefunc(tip, "SetHyperlink", function(self, link)
+        local id = type(link) == "string" and link:match("^spell:(%d+)")
+        if id then TrackSpell(tonumber(id)) end
+    end)
+end
+tip:HookScript("OnUpdate", function(self, elapsed)
+    if not query then return end
+    elapsedSinceUpdate = (elapsedSinceUpdate or 0) + elapsed
+    if elapsedSinceUpdate < 0.1 then return end
+    elapsedSinceUpdate = 0
+    Refresh()
+end)
